ограничения UNIQUE на пару полей (movie_session_id, ticket_id). Вместо full scan про джоине tickets и
booked_tickets, используется index scan по полю booked_tickets.ticket_id, взятое из unique ограничения на поля
(ticket_id, customer_id) таблицы booked_tickets.

CASE
WHEN bt.booked_ticket_id IS NULL THEN 'Свободно'
ELSE 'Занято'
END AS place_status
FROM movies AS m
INNER JOIN movie_sessions AS ms ON m.movie_id = ms.movie_id
INNER JOIN tickets AS t ON ms.movie_session_id = t.movie_session_id
LEFT JOIN booked_tickets AS bt ON t.ticket_id = bt.ticket_id
INNER JOIN places AS p ON t.place_id = p.place_id
INNER JOIN hall_rows AS hr ON p.hall_row_id = hr.hall_row_id

WHERE ms.movie_session_id = 1
ORDER BY hr.hall_row_number, p.place_number
)
